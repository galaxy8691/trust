**Crate 依赖关系：**

```
trustc ──────────── 依赖 ──── trust_parser, trust_hir, trust_tir, trust_codegen, trust_error
trust_codegen ──── 依赖 ──── trust_tir, trust_error
trust_tir ──────── 依赖 ──── trust_hir, trust_error
trust_hir ──────── 依赖 ──── trust_parser, trust_error
trust_parser ──── 依赖 ──── trust_error
trust_error ───── 依赖 ──── (零，基础 crate)
ferro_rt ──────── 依赖 ──── (零，Phase 1 无外部 crate)
```

> **规则：** 依赖方向必须是单向无环。`trust_error` 在最底层（被所有 crate 依赖），`trustc` 在最顶层。
> 
> **Cargo.lock 策略：** `trustc` 是 binary crate → 提交 `Cargo.lock`；其他 6 个 library crate → 不提交各自 `Cargo.lock`（由 workspace 根 `Cargo.lock` 统一管理，cargo 自动决定）。
> 
> **nightly 工具链说明：** CI 中 `coverage` 和 `miri` job 使用 nightly 工具链，仅用于分析工具（tarpaulin/miri），不用于编译器本身的编译。编译器代码始终通过 stable 1.63 验证。
> 
> **workspace MSRV 覆盖策略：** 若个别 crate 未来需要更高 MSRV（如 ferro_rt 引入 Tokio），可在其 `Cargo.toml` 中用 `[package] rust-version = "x.y"` 覆盖 workspace 默认值。

### 3.3 CI/CD 配置

#### 3.3.1 `.github/workflows/ci.yml` 完整规格

**触发条件：** PR 到 `main` 或 `phase1-compiler`；push 到 `main`。

```yaml
name: CI

on:
  pull_request:
    branches: [main, phase1-compiler]
  push:
    branches: [main]

env:
  CARGO_TERM_COLOR: always
  RUSTFLAGS: "-D warnings"

jobs:
  # ========== JOB 1: 基础检查（stable） ==========
  test-stable:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - run: cargo test --workspace
      - run: cargo build --workspace --release  # 确认 release 构建不失败

  # ========== JOB 2: Clippy + Fmt（stable） ==========
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
        with:
          components: clippy, rustfmt
      - run: cargo fmt --check --all
      - run: cargo clippy --workspace -- -D warnings -D clippy::unwrap_used -D clippy::expect_used -W clippy::cognitive_complexity=30 -W clippy::too_many_arguments=8

  # ========== JOB 3: Unsafe 检查（P0） ==========
  no-unsafe:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      # P0: trust_parser, trust_hir, trust_tir 中禁止 any unsafe
      - run: |
          for crate in trust_parser trust_hir trust_tir; do
            if grep -r "unsafe" "crates/$crate/src/" 2>/dev/null; then
              echo "ERROR: unsafe found in $crate (prohibited by constraints §3.2 P0)"
              exit 1
            fi
          done
          echo "OK: no unsafe in parser/hir/tir"

  # ========== JOB 4: Unsafe 注释审计（P1） ==========
  audit-unsafe:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      # P1: trust_codegen 和 ferro_rt 允许 unsafe，但每个块必须附带 // SAFETY: 注释
      - run: |
          for crate in trust_codegen ferro_rt; do
            if grep -rn "unsafe" "crates/$crate/src/" 2>/dev/null | grep -v "SAFETY" | grep -v "// SAFETY:" > /tmp/unsafe_audit.txt; then
              if [ -s /tmp/unsafe_audit.txt ]; then
                echo "ERROR: unsafe block without // SAFETY: comment found in $crate:"
                cat /tmp/unsafe_audit.txt
                exit 1
              fi
            fi
          done
          echo "OK: all unsafe blocks have // SAFETY: comments"

  # ========== JOB 5: audit-allow-unwrap（P0） ==========
  audit-allow-unwrap:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      # 每个 #[allow(clippy::unwrap_used)] 或 #[allow(clippy::expect_used)]
      # 上方 3 行内必须有 "// SAFETY:" 注释
      - run: |
          errors=0
          for file in $(grep -rl '#\[allow(clippy::unwrap_used)\|#\[allow(clippy::expect_used)' crates/ 2>/dev/null); do
            line=$(grep -n '#\[allow(clippy::unwrap_used)\|#\[allow(clippy::expect_used)' "$file" | head -1 | cut -d: -f1)
            start=$((line > 3 ? line - 3 : 1))
            if ! sed -n "${start},${line}p" "$file" | grep -q "// SAFETY:"; then
              echo "ERROR: $file:$line #[allow(unwrap/expect)] without // SAFETY: comment"
              errors=1
            fi
          done
          if [ $errors -eq 1 ]; then exit 1; fi
          echo "OK: all #[allow(unwrap/expect)] have // SAFETY: comments"

  # ========== JOB 6: 设计文档引用检查（P1） ==========
  doc-refs:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      # P0(P1): trust_tir/trust_codegen/trust_hir 中 pub fn 上方应有 // § 注释
      - run: |
          errors=0
          for crate in trust_tir trust_codegen trust_hir; do
            # 检查每个 pub fn 前 5 行内是否有 § 引用
            echo "Checking $crate..."
          done
          echo "NOTE: doc-refs full check active from Phase 1.3+"

  # ========== JOB 7: 覆盖率 — 工具检查（nightly） ==========
  coverage-tool:
    runs-on: ubuntu-latest
    continue-on-error: true  # 仅工具安装不稳定，继续不阻塞
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@nightly
      - name: Install tarpaulin
        run: cargo install cargo-tarpaulin
      - run: cargo tarpaulin --version

  # ========== JOB 8: 覆盖率 — 门控检查（nightly） ==========
  coverage-gate:
    runs-on: ubuntu-latest
    needs: coverage-tool
    # 不 continue-on-error：覆盖率不达标必须阻塞 merge
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@nightly
      - name: Install tarpaulin
        run: cargo install cargo-tarpaulin
      - name: trust_tir (≥85%)
        run: cargo tarpaulin -p trust_tir --fail-under 85 --out xml
      - name: trust_parser (≥70%)
        run: cargo tarpaulin -p trust_parser --fail-under 70 --out xml
      - name: trust_hir (≥70%)
        run: cargo tarpaulin -p trust_hir --fail-under 70 --out xml
      - name: trust_codegen (≥70%)
        run: cargo tarpaulin -p trust_codegen --fail-under 70 --out xml

  # ========== JOB 9: Miri unsafe 验证（nightly，P1） ==========
  miri:
    runs-on: ubuntu-latest
    continue-on-error: true
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@nightly
        with:
          components: miri
      - run: cargo miri test -p ferro_rt

  # ========== JOB 10: MSRV 测试（P1） ==========
  msrv:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@master
        with:
          toolchain: "1.63"
      - run: cargo test --workspace
```

**验收标准：**
